6. [Код и надежность]

```python
import re

def parse_power(value: str) -> int:
    match = re.match(r'^([\d.]+)\s*(kw|w)$', value.strip().lower())
    if not match:
        raise ValueError(f"Invalid format: {value}")
    num = float(match.group(1))
    return int(num * 1000 if match.group(2) == 'kw' else num)

# Примеры вызова
parse_power("450W")      # 450
parse_power("1.2 kW")    # 1200
parse_power("1500 w")    # 1500
parse_power("abc")       # ValueError
```
